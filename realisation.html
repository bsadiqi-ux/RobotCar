4. Programmation Arduino

Voici un extrait de programme permettant de mesurer la distance entre le robot et un obstacle grâce au capteur ultrasonique. 

// Déclaration des broches du capteur
const int trigPin = 9;
const int echoPin = 10;

long duree;
float distance;

// Initialisation du capteur
void setup() {

    pinMode(trigPin, OUTPUT);
    pinMode(echoPin, INPUT);

    // Démarrage de la communication série
    Serial.begin(9600);

}

// Mesure de la distance en boucle
void loop() {

    // Envoi d'une impulsion ultrasonique
    digitalWrite(trigPin, LOW);
    delayMicroseconds(2);

    digitalWrite(trigPin, HIGH);
    delayMicroseconds(10);

    digitalWrite(trigPin, LOW);

    // Lecture du signal de retour
    duree = pulseIn(echoPin, HIGH);

    // Calcul de la distance en centimètres
    distance = duree * 0.0343 / 2;

    // Affichage de la distance
    Serial.print("Distance : ");
    Serial.print(distance);
    Serial.println(" cm");

    delay(200);

}
